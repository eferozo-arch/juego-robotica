        let nivelActual = 0;
        let puntuacion = 0;
        let respondido = false;
        let posX = 20, posY = 310;
        const velocidad = 36;
        const robot = document.getElementById("robot");
        const zonaJuego = document.getElementById("zonaJuego");
        const pantallaPregunta = document.getElementById("pantallaPregunta");
        const pantallaFinal = document.getElementById("pantallaFinal");
        const etiquetaNivel = document.getElementById("etiquetaNivel");
        const textoPregunta = document.getElementById("textoPregunta");
        const contenedorOpciones = document.getElementById("contenedorOpciones");
        const retroalimentacion = document.getElementById("retroalimentacion");
        const btnSiguiente = document.getElementById("btnSiguiente");
        const puntuacionFinal = document.getElementById("puntuacionFinal");

        function cargarNivel() {
            respondido = false;
            const nivel = niveles[nivelActual];
            etiquetaNivel.textContent = nivel.etiqueta + " — Reto " + (nivelActual + 1) + " de " + niveles.length;
            
            posX = 20; posY = 310;
            actualizarPosicionRobot();

            document.querySelectorAll(".meta, .obstaculo").forEach(el => el.remove());

            const meta = document.createElement("div");
            meta.className = "meta";
            meta.style.left = nivel.meta.x + "px";
            meta.style.top = nivel.meta.y + "px";
            meta.textContent = "🎯";
            zonaJuego.appendChild(meta);

            nivel.obstaculos.forEach(obs => {
                const o = document.createElement("div");
                o.className = "obstaculo";
                o.style.left = obs.x + "px";
                o.style.top = obs.y + "px";
                o.style.width = obs.ancho + "px";
                o.style.height = obs.alto + "px";
                zonaJuego.appendChild(o);
            });

            pantallaPregunta.style.display = "none";
            pantallaFinal.style.display = "none";
            retroalimentacion.style.display = "none";
            btnSiguiente.style.display = "none";
        }

        function actualizarPosicionRobot() {
            robot.style.left = posX + "px";
            robot.style.top = posY + "px";
        }

        function choca(x, y) {
            const nivel = niveles[nivelActual];
            const tam = 44;
            for (const obs of nivel.obstaculos) {
                if (x < obs.x + obs.ancho && x + tam > obs.x &&
                    y < obs.y + obs.alto && y + tam > obs.y) return true;
            }
            return false;
        }

        function llegoMeta(x, y) {
            const nivel = niveles[nivelActual];
            const tam = 44;
            const metaTam = 48;
            return x < nivel.meta.x + metaTam && x + tam > nivel.meta.x &&
                   y < nivel.meta.y + metaTam && y + tam > nivel.meta.y;
        }

        function mostrarPregunta() {
            const nivel = niveles[nivelActual];
            pantallaPregunta.style.display = "block";
            textoPregunta.textContent = nivel.pregunta;
            contenedorOpciones.innerHTML = "";
            retroalimentacion.style.display = "none";
            btnSiguiente.style.display = "none";

            nivel.opciones.forEach((op, i) => {
                const btn = document.createElement("button");
                btn.className = "opcion";
                btn.textContent = op;
                btn.onclick = () => verificarRespuesta(i);
                contenedorOpciones.appendChild(btn);
            });
        }

        function verificarRespuesta(indice) {
            if (respondido) return;
            respondido = true;

            const nivel = niveles[nivelActual];
            const botones = document.querySelectorAll(".opcion");
            botones.forEach(b => b.disabled = true);

            if (indice === nivel.correcta) {
                puntuacion++;
                botones[indice].classList.add("correcta");
                retroalimentacion.className = "retroalimentacion retro-correcta";
                retroalimentacion.textContent = nivel.retroCorrecta;
                retroalimentacion.style.display = "block";
                btnSiguiente.style.display = "inline-block";
            } else {
                botones[indice].classList.add("incorrecta");
                botones[nivel.correcta].classList.add("correcta");
                retroalimentacion.className = "retroalimentacion retro-incorrecta";
                retroalimentacion.textContent = nivel.retroIncorrecta;
                retroalimentacion.style.display = "block";
                setTimeout(() => {
                    btnSiguiente.textContent = "Reintentar nivel";
                    btnSiguiente.style.display = "inline-block";
                }, 800);
            }
        }

        btnSiguiente.onclick = () => {
            const botonCorrecto = Array.from(document.querySelectorAll(".opcion")).find(b => 
                b.classList.contains("correcta") && !b.classList.contains("incorrecta")
            );
            if (botonCorrecto) {
                nivelActual++;
            }
            if (nivelActual < niveles.length) {
                cargarNivel();
            } else {
                zonaJuego.style.display = "none";
                pantallaPregunta.style.display = "none";
                pantallaFinal.style.display = "block";
                puntuacionFinal.textContent = `Obtuviste ${puntuacion} de ${niveles.length} respuestas correctas. ¡Sigue así! 🎓`;
            }
        };

        function reiniciar() {
            nivelActual = 0;
            puntuacion = 0;
            zonaJuego.style.display = "block";
            cargarNivel();
        }

        document.addEventListener("keydown", (e) => {
            if (pantallaPregunta.style.display === "block") return;

            let nx = posX, ny = posY;
            switch(e.key) {
                case "ArrowUp": ny -= velocidad; break;
                case "ArrowDown": ny += velocidad; break;
                case "ArrowLeft": nx -= velocidad; break;
                case "ArrowRight": nx += velocidad; break;
                default: return;
            }
            e.preventDefault();

            if (nx < 0 || nx > 520 - 44 || ny < 0 || ny > 380 - 44) return;
            if (choca(nx, ny)) return;

            posX = nx; posY = ny;
            actualizarPosicionRobot();

            if (llegoMeta(posX, posY)) {
                setTimeout(mostrarPregunta, 250);
            }
        });

        cargarNivel();
    </script>
